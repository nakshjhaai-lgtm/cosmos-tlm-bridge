# OpenC3 COSMOS v5 telemetry definition for the Turion DROID_SIM target.
# Consumed by both a real COSMOS instance (TargetModel) and cosmos-tlm-bridge's
# definition loader. Values are big-endian CCSDS-style packets.
#
# Conventions match OpenC3 Core + ingenium-lib-cosmos:
#   - EU (engineering units) are the CONVERTED values produced by POLY_READ_CONVERSION.
#   - DN (data numbers) are the RAW 16-bit counts on the wire.
#   - LIMITS are declared on the CONVERTED (EU) domain, as COSMOS does.
#     LIMITS <SET> <PERSISTENCE> <ENABLED|DISABLED> <RED_LOW> <YELLOW_LOW>
#            <YELLOW_HIGH> <RED_HIGH> [<GREEN_LOW> <GREEN_HIGH>]
#     GREEN_LOW/GREEN_HIGH define the optional BLUE "operational" band.

TELEMETRY DROID_SIM ADCS BIG_ENDIAN "Attitude determination & control subsystem"
  APPEND_ID_ITEM CCSDS_STREAMID 16 UINT 0x08AD
  APPEND_ITEM CCSDS_SEQUENCE 16 UINT "CCSDS packet sequence count"
  APPEND_ITEM CCSDS_LENGTH 16 UINT "CCSDS packet data length"
  APPEND_ITEM CCSDS_SECONDS 32 UINT "Coarse packet time, seconds since epoch"
  APPEND_ITEM CCSDS_SUBSECONDS 32 UINT "Fine packet time, 1/2^32 s"
  APPEND_ITEM ATTITUDE_ERROR 16 UINT "Attitude estimation error"
    UNITS Degrees deg
    FORMAT_STRING "%0.4f"
    POLY_READ_CONVERSION 0.0 0.001
    LIMITS DEFAULT 3 ENABLED -10.0 -5.0 5.0 10.0 -2.0 2.0
  APPEND_ITEM SLEW_ACTIVE 8 UINT "Slew maneuver in progress"
    STATE FALSE 0
    STATE TRUE 1
  APPEND_ITEM SLEW_TARGET_ID 8 UINT "Active slew target slot"
  APPEND_ITEM RW_SPEED 16 UINT "Reaction wheel speed magnitude"
    UNITS RPM rpm
    FORMAT_STRING "%0.1f"
    POLY_READ_CONVERSION 0.0 0.1
    LIMITS DEFAULT 3 ENABLED -6000.0 -5000.0 5000.0 6000.0

TELEMETRY DROID_SIM EPS BIG_ENDIAN "Electrical power subsystem"
  APPEND_ID_ITEM CCSDS_STREAMID 16 UINT 0x08E1
  APPEND_ITEM CCSDS_SEQUENCE 16 UINT
  APPEND_ITEM CCSDS_LENGTH 16 UINT
  APPEND_ITEM CCSDS_SECONDS 32 UINT
  APPEND_ITEM CCSDS_SUBSECONDS 32 UINT
  APPEND_ITEM BATT_VOLTAGE 16 UINT "Battery bus voltage"
    UNITS Volts V
    FORMAT_STRING "%0.3f"
    POLY_READ_CONVERSION 13.6 0.0025
    LIMITS DEFAULT 3 ENABLED 13.9 14.4 16.6 17.1 15.0 16.0
  APPEND_ITEM BATT_CURRENT 16 UINT "Battery bus current, signed via offset"
    UNITS Amps A
    FORMAT_STRING "%0.2f"
    POLY_READ_CONVERSION -8.0 0.01
    LIMITS DEFAULT 3 ENABLED -10.0 -6.0 2.0 6.0
  APPEND_ITEM BATT_SOC 16 UINT "Battery state of charge"
    UNITS Percent pct
    FORMAT_STRING "%0.1f"
    POLY_READ_CONVERSION 0.0 0.01
    LIMITS DEFAULT 3 ENABLED 5.0 15.0 98.0 100.0 40.0 90.0
  APPEND_ITEM BATT_TEMP 16 UINT "Battery cell temperature"
    UNITS Degrees_C C
    FORMAT_STRING "%0.1f"
    POLY_READ_CONVERSION -40.0 0.01
    LIMITS DEFAULT 3 ENABLED -20.0 0.0 40.0 55.0 5.0 35.0

    UNITS Percent pct
    FORMAT_STRING "%0.1f"
    POLY_READ_CONVERSION 0.0 0.01
